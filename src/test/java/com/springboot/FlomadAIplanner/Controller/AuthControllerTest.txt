        loginRequest = new LoginRequest(); // Initialize with test data as needed

    @Test
    void signUp_UserAlreadyExists() throws UserAlreadyExistsException {
        // Arrange
        doThrow(new UserAlreadyExistsException("User already exists")).when(userValidator).validateUserDoesNotExist(any(UserDTO.class));

        // Act
        ResponseEntity<String> response = authController.signUp(userDTO);

        // Assert
        assertEquals(HttpStatus.CONFLICT, response.getStatusCode());
        assertEquals("Sign-up failed: User already exists", response.getBody());
    }

    @Test
    void verifyUser_Success() {
        // Arrange
        String token = "valid_token";
        when(authService.verifyUser(token)).thenReturn(ResponseEntity.ok("Email verified successfully."));

        // Act
        ResponseEntity<String> response = authController.verifyUser(token);

        // Assert
        assertEquals(HttpStatus.OK, response.getStatusCode());
        assertEquals("Email verified successfully.", response.getBody());
    }

    @Test
    void verifyUser_InvalidToken() {
        // Arrange
        String token = "invalid_token";
        when(authService.verifyUser(token)).thenReturn(ResponseEntity.badRequest().body("Invalid verification token."));

        // Act
        ResponseEntity<String> response = authController.verifyUser(token);

        // Assert
        assertEquals(HttpStatus.BAD_REQUEST, response.getStatusCode());
        assertEquals("Invalid verification token.", response.getBody());
    }

    @Test
    void login_Success() {
        // Arrange
        LoginResponse expectedResponse = new LoginResponse(true, "dummy_jwt_token", "Login successful");
        when(authService.login(loginRequest)).thenReturn(ResponseEntity.ok(expectedResponse));

        // Act
        ResponseEntity<LoginResponse> response = authController.login(loginRequest);

        // Assert
        assertEquals(HttpStatus.OK, response.getStatusCode());
        assertEquals(expectedResponse, response.getBody());
    }

    @Test
    void login_Failure() {
        // Arrange
        LoginResponse expectedResponse = new LoginResponse(false, null, "Login failed: User not found");
        when(authService.login(loginRequest)).thenReturn(ResponseEntity.status(HttpStatus.UNAUTHORIZED).body(expectedResponse));

        // Act
        ResponseEntity<LoginResponse> response = authController.login(loginRequest);

        // Assert
        assertEquals(HttpStatus.UNAUTHORIZED, response.getStatusCode());
        assertEquals(expectedResponse, response.getBody());
    }
}
