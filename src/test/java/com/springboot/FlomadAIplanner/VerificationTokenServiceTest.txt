package com.springboot.FlomadAIplanner;

import org.junit.jupiter.api.Test;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.boot.test.context.SpringBootTest;
import org.springframework.util.Assert;

import com.springboot.FlomadAIplanner.model.entities.VerificationToken;
import com.springboot.FlomadAIplanner.service.model.VerificationTokenService;

@SpringBootTest
public class VerificationTokenServiceTest {

    @Autowired
    private VerificationTokenService verificationTokenService;

    @Test
    public void testSaveVerificationToken() {
        VerificationToken token = new VerificationToken();
        token.setId("1");
        token.setUserId("user1");

        VerificationToken savedToken = verificationTokenService.save(token);

        Assert.notNull(savedToken, "Saved verification token should not be null");
        Assert.isTrue(savedToken.getUserId().equals("user1"), "User ID should be 'user1'");
    }

    @Test
    public void testGetVerificationTokenById() {
        String id = "1";
        VerificationToken foundToken = verificationTokenService.getVerificationTokenById(id);

        Assert.notNull(foundToken, "Found verification token should not be null");
        Assert.isTrue(foundToken.getId().equals(id), "Token ID should match");
    }

    @Test
    public void testGetVerificationTokenByUserId() {
        String userId = "user1";
        VerificationToken foundToken = verificationTokenService.getVerificationTokenByUserId(userId);

        Assert.notNull(foundToken, "Found verification token should not be null");
        Assert.isTrue(foundToken.getUserId().equals(userId), "User ID should match");
    }

    @Test
    public void testDeleteVerificationTokenByUserId() {
        String userId = "user1";
        verificationTokenService.deleteVerificationTokenByUserId(userId);

        VerificationToken deletedToken = verificationTokenService.getVerificationTokenByUserId(userId);
        Assert.isNull(deletedToken, "Deleted verification token should be null");
    }
}
